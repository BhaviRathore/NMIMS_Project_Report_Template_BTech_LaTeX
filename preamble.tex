\usepackage[a4paper, left=1.5in, top=1in, right=1in, bottom=1in]{geometry}
\usepackage{fontspec}
\setmainfont{Times New Roman}

\usepackage{graphicx}

\usepackage{listings}
\lstset {
  basicstyle=\ttfamily,
  breaklines=true,
  columns=fullflexible
}

% used to force tables to stay in their section, with \begin{table}[H]
\usepackage{float}

% caption formatting
\usepackage[font=footnotesize, labelfont=bf]{caption}

% line space = 1.5
\usepackage{setspace}
\onehalfspacing

% headers and footers
\usepackage{fancyhdr}
% Redefine the 'plain' page style (used automatically by chapter pages)
\fancypagestyle{plain}{
  \fancyhf{} % clear defaults
  \fancyhead[L]{Project Title}
  \fancyhead[R]{Academic Year}  
  \fancyfoot[R]{\thepage} % puts the page number on the bottom right
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0pt}
}


\setlength{\headheight}{15pt}     % fixes warning

\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}



% headings format
\usepackage{titlesec}

% Pull the chapter title up to the top margin and add space below it
\titlespacing*{\chapter}{0pt}{-25pt}{20pt}

% Chapter title: 18 pt, bold
\titleformat{\chapter}[display]
  {\centering\bfseries\fontsize{18pt}{20pt}\selectfont}
  {Chapter \thechapter}
  {0pt}
  {}

% Section title: 14 pt, bold
\titleformat{\section}
  {\bfseries\fontsize{14pt}{16pt}\selectfont}
  {\thesection }{0.5em}{}

% Subsection title: 12 pt, bold
\titleformat{\subsection}
  {\bfseries\fontsize{12pt}{14pt}\selectfont}
  {\thesubsection }{0.5em}{}


% remove indent from first line of paragraph and add vertical spacing between paragraphs
\usepackage{parskip}
